from google import genai import streamlit as st # Настройка страницы st.set_page_config(page_title="AI Companion с Gemini", page_icon="✨") st.title("✨ Твой ИИ-компаньон") st.caption("Чат с реальной нейросетью Gemini") # Боковая панель для безопасного ввода API-ключа with st.sidebar: st.header("🔑 Настройки доступа") api_key = st.text_input( "Введи свой Gemini API Key", type="password", help="Ключ из Google AI Studio" ) st.markdown( "[Получить ключ бесплатно в Google AI" " Studio](https://aistudio.google.com/)" ) st.info( "Твой ключ никуда не передается и используется только для связи с" " официальным API." ) # Проверяем, введен ли ключ if not api_key: st.warning( "👈 Пожалуйста, введи свой API-ключ в боковой панели слева, чтобы чат" " заработал." ) else: # Инициализация клиента Google GenAI client = genai.Client(api_key=api_key) # Инициализация истории чата в сессии if "messages" not in st.session_state: st.session_state.messages = [ { "role": "assistant", "content": ( "Привет! Ключ успешно подключен. О чем бы тебе хотелось" " поболтать сегодня? 💫" ), } ] # Создаем или восстанавливаем сессию диалога для сохранения контекста if "chat_session" not in st.session_state: # Используем быструю и современную модель st.session_state.chat_session = client.chats.create( model="gemini-2.5-flash" ) # Отображение истории сообщений for message in st.session_state.messages: with st.chat_message(message["role"]): st.markdown(message["content"]) # Обработка ввода пользователя if prompt := st.chat_input("Напиши сообщение..."): # Добавляем сообщение пользователя в историю интерфейса st.session_state.messages.append({"role": "user", "content": prompt}) with st.chat_message("user"): st.markdown(prompt) # Отправка запроса к Gemini через официальный чат-клиент try: response = st.session_state.chat_session.send_message(prompt) reply = response.text except Exception as e: reply = ( "⚠️ Ошибка при обращении к API. Проверь правильность ключа." f" Детали:\n`{e}`" ) # Добавляем ответ ИИ в историю st.session_state.messages.append({"role": "assistant", "content": reply}) with st.chat_message("assistant"): st.markdown(reply)
